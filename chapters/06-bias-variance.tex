\chapter{The bias--variance decomposition}

Chapter~5 showed that no rule beats $\bar y$; this chapter measures how far a learned $h_S$ falls short of it, and why.

\begin{setup}
Population $F_{\vect{x},y}$ with regression function $\bar y(\vect{x})=\E_{y\given\vect{x}}(y)$; sample $S\sim F^n_{\vect{x},y}$ and learned rule $y=h_S(\vect{x})$; a new point $(\vect{x},y)\sim F_{\vect{x},y}$, \emph{independent of $S$}, to be predicted. Three things are random: $S$, $\vect{x}$ and $y$; every expectation names which it averages over.
\end{setup}

%=====================================================================
\section{Three sources of error}

\begin{definition}[Expected total error of prediction]
The \dfn{expected total error of prediction} of a learning algorithm is
\[
  \boxed{\;\text{ETEP}=\E_{\vect{x},y,S}\big[(h_S(\vect{x})-y)^2\big].\;}
\]
\end{definition}

Another sample gives another rule; averaging the rule over samples gives a fixed function.

\begin{definition}[Average predictor; Variance, Bias, Noise]
The \dfn{average predictor} is $\bar h(\vect{x})=\E_S\big[h_S(\vect{x})\big]$. With it,
\[
  \boxed{\;\begin{aligned}
    \text{Variance}&=\E_{\vect{x},S}\big[(h_S(\vect{x})-\bar h(\vect{x}))^2\big],\\
    \text{Bias}^2&=\E_{\vect{x}}\big[(\bar h(\vect{x})-\bar y(\vect{x}))^2\big],\\
    \text{Noise}&=\E_{\vect{x},y}\big[(y-\bar y(\vect{x}))^2\big].
  \end{aligned}\;}
\]
\end{definition}

What each term measures.
\begin{itemize}
  \item \emph{Variance:} how much the learned rule moves when the sample changes; high means the algorithm is very sensitive to the sample.
  \item \emph{Bias$^2$:} how far the average rule is from the truth $\bar y$; high means the algorithm searches an area away from the true one.
  \item \emph{Noise:} how much $y$ varies around $\bar y(\vect{x})$; it belongs to the population, does not involve $S$, and cannot be removed.
\end{itemize}

%=====================================================================
\section{The decomposition}

\begin{theorem}[Bias--variance decomposition]
\[
  \boxed{\;
  \begin{aligned}
  \underbrace{\E_{\vect{x},y,S}\big[(h_S(\vect{x})-y)^2\big]}_{\text{ETEP}}
  &=\underbrace{\E_{\vect{x},S}\big[(h_S(\vect{x})-\bar h(\vect{x}))^2\big]}_{\text{Variance}}
  +\underbrace{\E_{\vect{x}}\big[(\bar h(\vect{x})-\bar y(\vect{x}))^2\big]}_{\text{Bias}^2}\\
  &\quad+\underbrace{\E_{\vect{x},y}\big[(y-\bar y(\vect{x}))^2\big]}_{\text{Noise}}.
  \end{aligned}\;}
\]
\end{theorem}
\begin{proof}
Write $h_S$, $\bar h$, $\bar y$ for their values at $\vect{x}$.
\begin{steps}
  \item \emph{Add and subtract $\bar h$; expand.}
  \begin{deriv}
  \dl{\E_{\vect{x},y,S}\big[(h_S-y)^2\big]}{ETEP}
  \dl{\quad=\E_{\vect{x},y,S}\big[\big((h_S-\bar h)+(\bar h-y)\big)^2\big]}{add and subtract $\bar h$}
  \dl{\quad=\E_{\vect{x},S}\big[(h_S-\bar h)^2\big]+\E_{\vect{x},y}\big[(\bar h-y)^2\big]+2\,\text{MT}_1}{$(a+b)^2=a^2+b^2+2ab$; $y$ is absent from the first square, $S$ from the second}
  \end{deriv}
  with $\text{MT}_1=\E_{\vect{x},y,S}\big[(h_S-\bar h)(\bar h-y)\big]$.
  \item \emph{Claim: $\text{MT}_1=0$.}
  \begin{deriv}
  \dl{\text{MT}_1=\E_{\vect{x},y}\Big[\E_S\big((h_S-\bar h)(\bar h-y)\big)\Big]}{$S$ is independent of $(\vect{x},y)$: average over $S$ first, $(\vect{x},y)$ held fixed}
  \dl{\quad=\E_{\vect{x},y}\Big[\underbrace{(\bar h-y)}_{\text{independent of }S}\,\E_S\big(h_S-\bar h\big)\Big]}{pull out what is fixed given $(\vect{x},y)$}
  \dl{\quad=\E_{\vect{x},y}\big[(\bar h-y)\,(\bar h-\bar h)\big]=0}{$\E_S[h_S(\vect{x})]=\bar h(\vect{x})$}
  \end{deriv}
  \item \emph{Add and subtract $\bar y$ in the second term; expand.}
  \begin{deriv}
  \dl{\E_{\vect{x},y}\big[(\bar h-y)^2\big]=\E_{\vect{x},y}\big[\big((\bar h-\bar y)+(\bar y-y)\big)^2\big]}{add and subtract $\bar y$}
  \dl{\quad=\E_{\vect{x}}\big[(\bar h-\bar y)^2\big]+\E_{\vect{x},y}\big[(y-\bar y)^2\big]+2\,\text{MT}_2}{$(a+b)^2=a^2+b^2+2ab$; $(\bar h-\bar y)^2$ is free of $y$}
  \end{deriv}
  with $\text{MT}_2=\E_{\vect{x},y}\big[(\bar h-\bar y)(\bar y-y)\big]$.
  \item \emph{Claim: $\text{MT}_2=0$.}
  \begin{deriv}
  \dl{\text{MT}_2=\E_{\vect{x}}\Big[\underbrace{(\bar h-\bar y)}_{\text{free of }y}\,\E_{y\given\vect{x}}\big(\bar y-y\big)\Big]}{tower rule: $\E_{\vect{x},y}[\cdot]=\E_{\vect{x}}\big[\E_{y\given\vect{x}}(\cdot)\big]$; pull out}
  \dl{\quad=\E_{\vect{x}}\big[(\bar h-\bar y)\,(\bar y-\bar y)\big]=0}{$\E_{y\given\vect{x}}(y)=\bar y(\vect{x})$}
  \end{deriv}
  \item \emph{Assemble.} Steps 1--4 give ETEP $=$ Variance $+$ Bias$^2$ $+$ Noise.\qedhere
\end{steps}
\end{proof}

The same move twice: add and subtract the right average, and the middle term dies because a deviation from its own mean averages to zero. First over $S$ ($h_S$ around $\bar h$: Variance), then over $y\given\vect{x}$ ($y$ around $\bar y$: Noise); what is left, $\bar h$ against $\bar y$, is Bias$^2$. It is the learning form of $\MSE=\text{Variance}+\text{Bias}^2$, where the target was a fixed $\theta$, with no Noise and no $\vect{x}$.

A wider search space can get closer to $\bar y$ (low bias) but follows the particular sample more closely (high variance); a narrow one gives stable rules that may all be wrong.

\section{The four cases}

Each example fixes the population, then computes $\bar y$, $\bar h$ and each term.

\begin{example}[Low bias, low variance]
$x\sim U[0,100]$, $y\given x\sim U[x,\,x+0.1]$; algorithm $h_S(x)=x+0.05$ (it ignores $S$).
\begin{deriv}
\dl{\bar y(x)=\E_{y\given x}(y)=\frac{x+(x+0.1)}{2}=x+0.05}{mean of $U[a,b]$ is $\frac{a+b}2$}
\dl{\bar h(x)=\E_S[x+0.05]=x+0.05}{$h_S$ does not depend on $S$}
\dl{\text{Bias}^2=\E_x\big[(\bar h-\bar y)^2\big]=\E_x[0]=0}{$\bar h=\bar y$}
\dl{\text{Variance}=\E_{x,S}\big[(h_S-\bar h)^2\big]=\E_{x,S}[0]=0}{$h_S=\bar h$}
\dl{\text{Noise}=\E_x\big[\Var_{y\given x}(y)\big]=\frac{0.1^2}{12}\approx0.00083}{tower rule; $U[a,b]$ variance $\frac{(b-a)^2}{12}$, the same at every $x$}
\end{deriv}
ETEP $=0.00083$: only the Noise is left.
\end{example}

\begin{example}[High bias, low variance]
Population as above; algorithm $h_S(x)=x+100.05$.
\begin{deriv}
\dl{\bar h(x)=x+100.05}{$h_S$ ignores $S$}
\dl{\text{Bias}^2=\E_x\big[(x+100.05-x-0.05)^2\big]=10^4}{$\bar y(x)=x+0.05$}
\dl{\text{Variance}=0,\quad\text{Noise}=0.00083}{$h_S=\bar h$; Noise depends only on the population}
\end{deriv}
Perfectly stable, and always wrong by $100$.
\end{example}

\begin{example}[Low bias, high variance]
$x\sim U[-100,100]$, $y\given x\sim N(1+x,\,0.01)$. Algorithm: take the slope as $1$, estimate the intercept by $\hat a=\bar y_S-\bar x_S$ (difference of the sample means of the $y_i$ and the $x_i$), then add the first sample point's $x_1$:
\[
  h_S(x)=\hat a+x+x_1.
\]
\begin{deriv}
\dl{\bar y(x)=1+x,\quad\text{Noise}=\E_x\big[\Var_{y\given x}(y)\big]=0.01}{$y\given x\sim N(1+x,0.01)$}
\dl{y_i-x_i=1+\varepsilon_i,\quad\varepsilon_i\iid N(0,0.01)}{$y=\bar y(x)+\varepsilon$; the $\varepsilon_i$ are independent of the $x_i$}
\dl{\hat a=1+\bar\varepsilon,\quad\E_S[\hat a]=1,\quad\Var_S(\hat a)=\tfrac{0.01}n}{average the $n$ rows above}
\dl{\bar h(x)=\E_S[\hat a]+x+\E_S[x_1]=1+x=\bar y(x)}{$\E_S[x_1]=0$ for $x_1\sim U[-100,100]$}
\dl{\text{Bias}^2=0}{$\bar h=\bar y$}
\dl{h_S(x)-\bar h(x)=\bar\varepsilon+x_1}{the same at every $x$; mean $0$}
\dl{\text{Variance}=\Var_S(\bar\varepsilon+x_1)}{$\E_{x,S}$ of a square that is the same at every $x$}
\dl{\quad=\tfrac{0.01}n+\tfrac{200^2}{12}\approx\tfrac{0.01}n+3333.3}{$\bar\varepsilon$ independent of $x_1$: variances add; $U[a,b]$ variance $\frac{(b-a)^2}{12}$}
\end{deriv}
Right on average, but each rule is shifted by a random $x_1$ of size up to $100$; the Variance never falls below $3333$, however large $n$ is.
\end{example}

\begin{example}[High bias, high variance]
Population and $h_S$ as in the last example; algorithm $h'_S(x)=h_S(x)+1000$.
\begin{deriv}
\dl{\bar h'(x)=\bar h(x)+1000=1+x+1000}{$\E_S$ of a shift}
\dl{\text{Bias}^2=\E_x\big[1000^2\big]=10^6}{$\bar y(x)=1+x$}
\dl{h'_S-\bar h'=h_S-\bar h\ \Rightarrow\ \text{Variance}\approx3333.3+\tfrac{0.01}n}{the shift cancels; Noise still $0.01$}
\end{deriv}
Wrong on average, and unstable too.
\end{example}
